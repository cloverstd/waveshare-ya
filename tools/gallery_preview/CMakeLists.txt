cmake_minimum_required(VERSION 3.16)
project(ya_gallery_preview C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(YA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${YA_ROOT}/tools/typography_preview/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${YA_ROOT}/managed_components/lvgl__lvgl" lvgl)
file(GLOB YA_FONTS "${YA_ROOT}/main/font_*.c")
add_executable(gallery_preview preview.cpp ${YA_FONTS} "${YA_ROOT}/main/gallery_app.cpp")
target_compile_definitions(gallery_preview PRIVATE YA_GALLERY_PREVIEW=1)
target_include_directories(gallery_preview PRIVATE "${YA_ROOT}/main" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gallery_preview PRIVATE lvgl)
